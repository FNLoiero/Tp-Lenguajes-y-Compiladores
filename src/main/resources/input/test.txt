#+ 2. continue sin etiqueta: el write no se ejecuta cuando i = 3. Imprime: 1 2 4 5 +#

        continue
    }

}

#+ 3. break con etiqueta: corta los DOS ciclos cuando i*j > 6. Imprime: 2 4 (valores de i y j) +#
write("break externo")
i := 0
j := 0
externo: while (i < 5)
{
    j := 0
    while (j < 5)
    {
        if (i * j > 6)
        {
            break externo
        }
        j := j + 1
    }
    i := i + 1
}
write(i)
write(j)

#+ 4. continue con etiqueta: vuelve a evaluar la condicion del ciclo externo. Imprime: 0 1 0 1 +#
write("continue externo")
i := 0
externo: while (i < 5)

    }
}
